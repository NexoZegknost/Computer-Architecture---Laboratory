\subsection{Exercise 2}
\task{Requirement}
Input an array with 5 elements (prompts: ``Please input element 0:'',
``Please input element 1:'', \dots), then print the element at the index chosen
by the user (0 to 4) with the prompt ``Please enter index:''.
For example, with the array \textbf{5, 10, 15, 20, 25}, input \textbf{1}
gives \textbf{10}, input \textbf{4} gives \textbf{25}.

\task{Test cases}
\testcase{2}{1}{elements \texttt{5, 10, 15, 20, 25}; index \texttt{1}}{\texttt{10}}
\testcase{2}{2}{elements \texttt{5, 10, 15, 20, 25}; index \texttt{4}}{\texttt{25}}
\testcase{2}{3}{elements \texttt{-7, 3, 0, 42, -100}; index \texttt{4}}{\texttt{-100}}
