\subsection{Exercise 3}
\task{Requirement}
Print out the result of
\[
  F = \frac{(a+10)\times(b-d)\times(c-2\times a)}{a+b+c}
\]
where $a$, $b$, $c$, $d$ are integers inserted in this exact order
(prompts: ``Insert a: '', ``Insert b: '', \dots), $a+b+c \neq 0$.
Both the quotient and the remainder must be printed.

\task{Test cases}
\testcase{3}{1}{$a=3$, $b=4$, $c=5$, $d=6$}{\texttt{F = 2, remainder 2}
  \hfill $\left(\frac{13 \times (-2) \times (-1)}{12} = \frac{26}{12}\right)$}
\testcase{3}{2}{$a=0$, $b=7$, $c=5$, $d=2$}{\texttt{F = 20, remainder 10}
  \hfill $\left(\frac{10 \times 5 \times 5}{12} = \frac{250}{12}\right)$}
\testcase{3}{3}{$a=1$, $b=2$, $c=3$, $d=4$}{\texttt{F = -3, remainder -4}
  \hfill $\left(\frac{11 \times (-2) \times 1}{6} = \frac{-22}{6}\right)$}
