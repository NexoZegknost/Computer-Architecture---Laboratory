\subsection{Exercise 3}
\task{Requirement}
Write a MIPS program to print out the result of F:
\[
  F = \frac{(a+10)\times(b-d)\times(c-2\times a)}{a+b+c}
\]
where a, b, c, and d are integers inserted by the user. Note that a, b, c,
and d are inserted in this exact order. Moreover, ``a+b+c'' is assume to not
equal to 0. The output of division should print both the quotient and the
remainder.\\
The input prompts should be as such: ``Insert a: '', ``Insert b: '', etc.\\
For example, if the input is 3, 4, 5, 6 then the output should be: \textbf{F = 2,
remainder 2}.

\task{Test cases}
\testcase{3}{1}{$a=3$, $b=4$, $c=5$, $d=6$}{\texttt{F = 2, remainder 2}
  \hfill $\left(\frac{13 \times (-2) \times (-1)}{12} = \frac{26}{12}\right)$}
\testcase{3}{2}{$a=0$, $b=7$, $c=5$, $d=2$}{\texttt{F = 20, remainder 10}
  \hfill $\left(\frac{10 \times 5 \times 5}{12} = \frac{250}{12}\right)$}
\testcase{3}{3}{$a=1$, $b=2$, $c=3$, $d=4$}{\texttt{F = -3, remainder -4}
  \hfill $\left(\frac{11 \times (-2) \times 1}{6} = \frac{-22}{6}\right)$}
